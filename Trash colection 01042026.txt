class OrnitoEnv(gym.Env):
    def __init__(self, model_path):
        super(OrnitoEnv, self).__init__()
        # Carrega o modelo XML que construímos
        self.model = mujoco.MjModel.from_xml_path(model_path)
        self.data = mujoco.MjData(self.model)
        
        # AÇÃO: 5 valores normalizados entre -1 e 1 (as 5 juntas) [cite: 82, 163]
        self.action_space = spaces.Box(low=-1, high=1, shape=(5,), dtype=np.float32)
        
        # OBSERVAÇÃO: Sensores + Histórico + Trajetória futura [cite: 167, 169]
        # (Definiremos o tamanho exato na próxima etapa)
        self.observation_space = spaces.Box(low=-np.inf, high=np.inf, shape=(120,), dtype=np.float32)

        self.f_c = 7.0  # Frequência de corte (Hz)
        self.dt = 0.02  # Passo da IA (50Hz)
        
        # Calculando o alpha (aproximação clássica)
        self.alpha = (2 * np.pi * self.dt * self.f_c) / (2 * np.pi * self.dt * self.f_c + 1) # Alpha é o coeficiente de suavização
        
        # Estado anterior do filtro (começa em zero para as 5 juntas)
        self.last_action = np.zeros(5)

        # Guarda as últimas 25 observações (0.5 segundos a 50Hz)
        self.history = deque(maxlen=25)

    def get_observation(self):
        # 1. Orientação (Quatérnio: 4 valores) [cite: 165]
        # No MuJoCo, o quatérnio da base livre está em data.qpos[3:7]
        eta = self.data.qpos[3:7] 

        # 2. Velocidade Angular Local (3 valores: p, q, r) [cite: 165]
        # data.qvel[3:6] nos dá a velocidade angular da base
        pqr = self.data.qvel[3:6]

        # 3. Posições das Juntas (5 valores: q1 a q5) [cite: 165]
        # Pegamos apenas as posições das 5 juntas que criamos (ignorando a base livre)
        qj = self.data.qpos[7:12]

        # 4. Velocidade do Ar (Pitot Tube - 1 valor) [cite: 165]
        # O artigo modela isso como a velocidade X local relativa ao ar 
        # Para simplificar agora: pegamos a velocidade linear X do torso
        vx_air = np.array([self.data.qvel[0]]) 

        # Concatenamos tudo em um único vetor (13 valores por enquanto)
        obs_atual = np.concatenate([eta, pqr, qj, vx_air])
        
        return obs_atual

    def compute_reward(self, action):
        # 1 Erro de Posição (r_pos)
        pos_atual = self.data.qpos[:3]
        pos_alvo = self.data.mocap_pos[0]
        distancia = np.linalg.norm(pos_atual - pos_alvo) # Distância entre o torso e o alvo
        
        r_pos = np.exp(-2.0 * (distancia**2)) # Usa-se uma curva de sino para que a recompensa seja próxima de 1 apenas quando o erro próximo de 0.

        # 2. Estabilidade Angular (r_omega)
        # Penaliza velocidades angulares altas (p, q, r) para evitar que o robô gire loucamente
        angular_vel = self.data.qvel[3:6]
        r_omega = np.exp(-0.1 * np.sum(np.square(angular_vel)))

        # 3. Estabilidade de Atitude (r_att)
        # O artigo penaliza desvios excessivos de inclinação (Roll/Pitch)
        # Aqui SIMPLIFICAMOS usando a diferença do quatérnio para a identidade (voo nivelado)
        quat = self.data.qpos[3:7] # [w, x, y, z]
        # Recompensa maior se o componente 'w' for próximo de 1 (ou seja, sem rotação)
        r_att = quat[0]**2

        # 4. Eficiência Energética (r_en)
        # Penaliza o uso de torques altos (as ações da IA)
        r_en = -np.sum(np.square(action))

        # Recompensa Final Ponderada
        reward = (0.5 * r_pos) + (0.1 * r_omega) + (0.2 * r_att) + (0.05 * r_en)
        
        return reward

    def apply_filter(self, action_bruta):
            # Aplica a equação: y = alpha*x + (1-alpha)*y_old
            action_filtrada = self.alpha * action_bruta + (1 - self.alpha) * self.last_action
            self.last_action = action_filtrada
            return action_filtrada

    def step(self, action):
